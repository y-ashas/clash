/* Full fixed main.js — unit height + proper tower targeting and bounds clamping. - Units spawn and bob at y = 0.6 (above ground) - Units acquire nearest alive enemy tower and retarget when towers die - Units are clamped inside ARENA rectangle (no walking out of map) - Minimal tower objects with hp and simple damage-on-contact */ let scene, camera, renderer, controls; let overlay2D, hpCanvas; let octx, hctx; let lastTime = performance.now(); const MAX_ELIXIR = 10; const ARENA = { width: 24, height: 36 }; // X, Z extents (centered at 0) const ARENA_HALF = { x: ARENA.width / 2, z: ARENA.height / 2 }; const STATE = { mode: null, started: false, gameOver: false, elapsed: 0, selectedLane: 0 }; const TROOP_BY_KEY = { knight: { key: 'knight', name: 'Knight', cost: 3, hp: 100, speed: 2.2 } }; const GAME = { elixir: { P1: 4, P2: 4 }, hands: { P1: ['knight','knight'], P2: ['knight'] }, units: [], towers: [] }; /* --------------------------- Arena / towers creation --------------------------- */ function setupArena() { // remove existing towers/models GAME.towers.forEach(t => { try { if (t.mesh) scene.remove(t.mesh); } catch (e){} }); GAME.towers = []; // offsets: place enemy towers near far end (-z) and player towers near +z const towerOffsetX = 6; const towerZPlayer = ARENA_HALF.z - 3; // player's princess towers (positive z) const towerZEnemy = -ARENA_HALF.z + 3; // enemy's princess towers (negative z) const kingZPlayer = ARENA_HALF.z; // king tower on ends (could be center) const kingZEnemy = -ARENA_HALF.z; // helper to make tower function makeTower(owner, type, x, z) { const t = { id: (GAME.towers.length+1), owner, type, hp: type==='king'? 300 : 120, pos: new THREE.Vector3(x, 0, z), alive: true, mesh: null }; const geo = new THREE.BoxGeometry(1.2, 2.2, 1.2); const mat = new THREE.MeshStandardMaterial({ color: owner==='P1'?0x4caf50:0xf44336 }); const mesh = new THREE.Mesh(geo, mat); mesh.position.set(x, 1.1, z); scene.add(mesh); t.mesh = mesh; GAME.towers.push(t); return t; } // Create P2 (enemy) towers makeTower('P2','princess', -towerOffsetX, towerZEnemy); makeTower('P2','princess', towerOffsetX, towerZEnemy); makeTower('P2','king', 0, kingZEnemy); // Create P1 (player) towers makeTower('P1','princess', -towerOffsetX, towerZPlayer); makeTower('P1','princess', towerOffsetX, towerZPlayer); makeTower('P1','king', 0, kingZPlayer); } /* --------------------------- Unit class: height + targeting --------------------------- */ class Unit { constructor(troop, owner, laneIndex) { this.id = (Unit._nextId = (Unit._nextId || 0) + 1); this.troop = troop; this.owner = owner; // 'P1' or 'P2' this.hp = troop.hp; // spawn a bit above ground so they don't intersect with the floor const spawnX = owner === 'P1' ? 4 : -4; const spawnZ = (laneIndex === 0 ? -2 : 2); this.pos = new THREE.Vector3(spawnX, 0.6, spawnZ); // y = 0.6 baseline this.speed = troop.speed; this.target = null; this.mesh = this._makeMesh(); if (scene) scene.add(this.mesh); } _makeMesh(){ const g = new THREE.BoxGeometry(0.7, 1.2, 0.7); const m = new THREE.MeshStandardMaterial({ color: this.owner === 'P1' ? 0x88cc88 : 0xcc8888 }); const mesh = new THREE.Mesh(g, m); mesh.position.copy(this.pos); return mesh; } alive(){ return this.hp > 0; } acquireTarget() { // find nearest alive enemy tower const enemies = GAME.towers.filter(t => t.alive && t.owner !== this.owner); if (enemies.length === 0) { this.target = null; return; } // prefer non-king towers first (so they attack princess towers), but will pick nearest enemies.sort((a,b) => { const da = a.type === 'king' ? 0.5 : 1.0; const db = b.type === 'king' ? 0.5 : 1.0; const distA = this.pos.distanceTo(a.pos) / da; const distB = this.pos.distanceTo(b.pos) / db; return distA - distB; }); this.target = enemies[0]; } update(dt) { // retarget if needed if (!this.target || !this.target.alive) this.acquireTarget(); if (!this.target) { // no target: slow idling and stay within bounds this._clampPosition(); if (this.mesh) this.mesh.position.copy(this.pos); return; } // move toward target position (slightly above ground) const to = new THREE.Vector3(this.target.pos.x, 0, this.target.pos.z); const dir = to.clone().sub(new THREE.Vector3(this.pos.x, 0, this.pos.z)); const dist = dir.length(); if (dist > 0.05) { dir.normalize(); const moveDist = Math.min(this.speed * dt, dist); this.pos.x += dir.x * moveDist; this.pos.z += dir.z * moveDist; } // clamp to arena so they don't roam out this._clampPosition(); // update mesh with raised y and bob if (this.mesh) { this.mesh.position.x = this.pos.x; this.mesh.position.z = this.pos.z; this.mesh.position.y = 0.6 + 0.02 * Math.sin(performance.now() / 150 + this.id); } // if close enough to attack the tower, deal damage over time if (dist < 1.1 && this.target && this.target.alive) { // simple DPS: reduce tower hp const damage = 20 * dt; // 20 HP per second this.target.hp -= damage; if (this.target.mesh) { // flash or scale down slightly to indicate damage this.target.mesh.scale.y = 1 - Math.min(0.6, (this.target.hp / (this.target.type==='king'?300:120))); } if (this.target.hp <= 0) { this.target.alive = false; // remove tower mesh try { if (this.target.mesh) scene.remove(this.target.mesh); } catch (e){} // force units to retarget on next update GAME.towers = GAME.towers.map(t => t.id === this.target.id ? this.target : t); this.target = null; } } } _clampPosition() { // keep units inside ARENA bounds with a small margin const margin = 0.5; this.pos.x = Math.max(-ARENA_HALF.x + margin, Math.min(ARENA_HALF.x - margin, this.pos.x)); this.pos.z = Math.max(-ARENA_HALF.z + margin, Math.min(ARENA_HALF.z - margin, this.pos.z)); } destroy() { try { if (this.mesh) scene.remove(this.mesh); } catch(e){} this.hp = 0; } } /* --------------------------- Game loop and systems --------------------------- */ function tickElixir(dt){ ['P1','P2'].forEach(p => GAME.elixir[p] = Math.min(MAX_ELIXIR, GAME.elixir[p] + dt * 0.5)); } function tickUnits(dt){ for (let i = GAME.units.length - 1; i >= 0; i--) { const u = GAME.units[i]; if (!u.alive()) { u.destroy(); GAME.units.splice(i,1); continue; } u.update(dt); } } function botProactive(dt){ if (STATE.mode === 'bot' && Math.random() < 0.01) { // spawn a unit for P2 if enough elixir const pick = TROOP_BY_KEY.knight; if (GAME.elixir.P2 >= pick.cost) { GAME.elixir.P2 -= pick.cost; const unit = new Unit(pick, 'P2', Math.random() < 0.5 ? 0 : 1); GAME.units.push(unit); } } } function updateElixirUI(){ const p1 = document.getElementById('p1ElixirFill'); if (p1) p1.style.width = Math.round((GAME.elixir.P1 / MAX_ELIXIR) * 100) + '%'; } /* --------------------------- Start game / rendering --------------------------- */ function initThree() { scene = new THREE.Scene(); scene.background = new THREE.Color('#80ccff'); camera = new THREE.PerspectiveCamera(60, window.innerWidth/window.innerHeight, 0.1, 500); camera.position.set(12, 14, 20); camera.lookAt(0, 0, 0); renderer = new THREE.WebGLRenderer({ antialias: true }); renderer.setSize(window.innerWidth, window.innerHeight); renderer.shadowMap.enabled = true; document.body.appendChild(renderer.domElement); controls = new THREE.OrbitControls(camera, renderer.domElement); controls.enableDamping = true; // ground as a thin box so units don't intersect const ground = new THREE.Mesh(new THREE.BoxGeometry(ARENA.width, 0.2, ARENA.height), new THREE.MeshStandardMaterial({ color: 0x153047 })); ground.position.y = -0.1; scene.add(ground); // lights const hemi = new THREE.HemisphereLight(0xffffff, 0x333344, 0.7); scene.add(hemi); const dir = new THREE.DirectionalLight(0xffffff, 0.9); dir.position.set(10,18,8); scene.add(dir); setupArena(); } function startGame() { if (STATE.started) return; STATE.started = true; initThree(); lastTime = performance.now(); requestAnimationFrame(gameLoop); } function gameLoop(now) { const dt = Math.min(0.05, (now - lastTime) / 1000); lastTime = now; if (STATE.started && !STATE.gameOver) { tickElixir(dt); tickUnits(dt); botProactive(dt); updateElixirUI(); } if (controls) controls.update(); if (renderer && scene && camera) renderer.render(scene, camera); requestAnimationFrame(gameLoop); } /* Expose for index.html */ window.selectMode = function(mode){ STATE.mode = mode; startGame(); }; window.startGame = startGame; console.log('main.js loaded (units target towers and are clamped inside the arena).');
